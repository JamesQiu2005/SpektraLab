cmake_minimum_required(VERSION 3.20)
project(SpektraInputDependencies LANGUAGES C CXX)
include(ExternalProject)
if(NOT SPEKTRALAB_INPUT_SOURCES)
    message(FATAL_ERROR "Pass SPEKTRALAB_INPUT_SOURCES, populated by setup-input-codecs.py")
endif()
set(common -DCMAKE_BUILD_TYPE=Release -DCMAKE_INSTALL_PREFIX=${CMAKE_INSTALL_PREFIX}
    -DCMAKE_INSTALL_LIBDIR=lib -DCMAKE_C_COMPILER=${CMAKE_C_COMPILER}
    -DCMAKE_CXX_COMPILER=${CMAKE_CXX_COMPILER} -DCMAKE_MAKE_PROGRAM=${CMAKE_MAKE_PROGRAM}
    -DBUILD_SHARED_LIBS=OFF -DBUILD_TESTING=OFF -DCMAKE_POSITION_INDEPENDENT_CODE=ON)
if(CMAKE_TOOLCHAIN_FILE)
    list(APPEND common -DCMAKE_TOOLCHAIN_FILE=${CMAKE_TOOLCHAIN_FILE})
endif()
ExternalProject_Add(input_imath SOURCE_DIR "${SPEKTRALAB_INPUT_SOURCES}/Imath-3.2.3"
    DOWNLOAD_COMMAND "" UPDATE_COMMAND "" CMAKE_ARGS ${common} -DPYTHON=OFF -DPYBIND11=OFF
    LOG_CONFIGURE ON LOG_BUILD ON LOG_INSTALL ON)
ExternalProject_Add(input_exr SOURCE_DIR "${SPEKTRALAB_INPUT_SOURCES}/openexr-3.5.2"
    DEPENDS input_imath DOWNLOAD_COMMAND "" UPDATE_COMMAND ""
    CMAKE_ARGS ${common} -DCMAKE_PREFIX_PATH=${CMAKE_INSTALL_PREFIX}
        -DOPENEXR_BUILD_TOOLS=OFF -DOPENEXR_BUILD_EXAMPLES=OFF -DOPENEXR_BUILD_PYTHON=OFF
        -DOPENEXR_FORCE_INTERNAL_DEFLATE=ON -DOPENEXR_FORCE_INTERNAL_OPENJPH=ON -DOPENEXR_FORCE_INTERNAL_ZSTD=ON
    LOG_CONFIGURE ON LOG_BUILD ON LOG_INSTALL ON)
ExternalProject_Add(input_de265 SOURCE_DIR "${SPEKTRALAB_INPUT_SOURCES}/libde265-1.1.3"
    DOWNLOAD_COMMAND "" UPDATE_COMMAND "" CMAKE_ARGS ${common}
        -DENABLE_DECODER=OFF -DENABLE_ENCODER=OFF -DENABLE_SDL=OFF -DENABLE_AVX512=OFF
    LOG_CONFIGURE ON LOG_BUILD ON LOG_INSTALL ON)
ExternalProject_Add(input_heif SOURCE_DIR "${SPEKTRALAB_INPUT_SOURCES}/libheif-1.23.6"
    DEPENDS input_de265 DOWNLOAD_COMMAND "" UPDATE_COMMAND "" CMAKE_ARGS ${common}
        -DCMAKE_PREFIX_PATH=${CMAKE_INSTALL_PREFIX} -DCMAKE_CXX_FLAGS=-DLIBDE265_STATIC_BUILD=1
        -DWITH_LIBDE265=ON -DWITH_LIBDE265_PLUGIN=OFF -DENABLE_PLUGIN_LOADING=OFF
        -DWITH_X265=OFF -DWITH_X264=OFF -DWITH_OpenH264_DECODER=OFF
        -DWITH_AOM_DECODER=OFF -DWITH_AOM_ENCODER=OFF -DWITH_LIBSHARPYUV=OFF
        -DWITH_EXAMPLES=OFF -DWITH_GDK_PIXBUF=OFF -DBUILD_DOCUMENTATION=OFF
        -DENABLE_PARALLEL_TILE_DECODING=OFF
    LOG_CONFIGURE ON LOG_BUILD ON LOG_INSTALL ON)
file(GLOB lcms_sources CONFIGURE_DEPENDS "${SPEKTRALAB_INPUT_SOURCES}/Little-CMS-lcms2.19.1/src/*.c")
add_library(spk_lcms STATIC ${lcms_sources})
target_include_directories(spk_lcms PUBLIC "${SPEKTRALAB_INPUT_SOURCES}/Little-CMS-lcms2.19.1/include")
set_target_properties(spk_lcms PROPERTIES OUTPUT_NAME lcms2 POSITION_INDEPENDENT_CODE ON)
install(TARGETS spk_lcms ARCHIVE DESTINATION lib)
install(FILES "${SPEKTRALAB_INPUT_SOURCES}/Little-CMS-lcms2.19.1/include/lcms2.h"
              "${SPEKTRALAB_INPUT_SOURCES}/Little-CMS-lcms2.19.1/include/lcms2_plugin.h" DESTINATION include)
foreach(notice "libheif-1.23.6/COPYING" "libde265-1.1.3/COPYING" "libde265-1.1.3/libde265/COPYING"
    "openexr-3.5.2/LICENSE.md" "openexr-3.5.2/PATENTS" "openexr-3.5.2/external/deflate/COPYING"
    "openexr-3.5.2/external/OpenJPH/LICENSE" "openexr-3.5.2/external/zstd/LICENSE"
    "Imath-3.2.3/LICENSE.md" "Little-CMS-lcms2.19.1/LICENSE")
    string(REPLACE "/" "-" name "${notice}")
    install(FILES "${SPEKTRALAB_INPUT_SOURCES}/${notice}" DESTINATION share/spektralab-input-codecs/licenses RENAME "${name}")
endforeach()
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../input-codecs-pins.json" DESTINATION share/spektralab-input-codecs)
install(FILES "${SPEKTRALAB_INPUT_SOURCES}/libheif-1.23.6/tests/data/with-alpha-512x512.heic"
    DESTINATION share/spektralab-input-codecs/tests)
